ApplicationInfo#

应用信息。

class ApplicationInfo {
    var iconPath: String? = null
    var versionName: String? = null
    var packageName: String? = null
    var appName: String? = null
}

Property#

String iconPath : 应用icon文件路径。

String versionName : 应用版本号。

String packageName : 应用包名。

String appName : 应用名称。
