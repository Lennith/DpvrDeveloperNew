WifiInfo#

Wifi information structure.

class WifiInfo{
    var SSID: String
    var cipherType = WifiManagerModule.WIFI_CIPHER_WPA
    var frequency = WifiManagerModule.WIFI_FREQUENCY_2400
    var ip = "0.0.0.0"
    var level = 0
    var state = NetworkInfo.State.UNKNOWN.name
}

Property#

String SSID : The SSID of the Wifi.

String cipherType:Wifi encryption type “WEP”: WEP encryption.

“WPA”: WPA encryption.

“NO_CIPHER”: Unencrypted.

String frequency: Wifi frequency band “2.4g” : 2.4g band.

“5g” : 5g band.

String ip: IP address, only valid after successful connection.

int level : Wifi strength.

String state : Current status.

“CONNECTING” : Connecting.

“CONNECTED” : Connected.

“SUSPENDED” : Suspended.

“DISCONNECTING” : Disconnecting.

“DISCONNECTED” : Disconnected.

“UNKNOWN” : Unknown.
