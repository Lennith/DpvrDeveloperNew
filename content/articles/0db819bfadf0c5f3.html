RemoteDeviceControl#

Here is an example of the usage of an RDC interface, which periodically sends a message and prints the received server information after detecting a connection established between the RDC and the server.

class ApiTest {
companion object RDCTestStatic {
    const val TAG = "DMST-RDC"
    const val EVENT_NAME_SERVER_MESSAGE = "rcs_server_message"
}

private var mIsConnected = false
private var messageCount = 0


inner class CheckConnectionResult {
    var isConnected: Boolean = false
}

inner class SendServerMessageParams {
    var message: String? = null
}

inner class RCSServerMessageEvent {
    @Expose
    var message: String? = null
}

private val mServerMessageReceivedHandler = object : TestDeviceManager.EventCallback {
    override fun onEvent(eventInfo: String?) {
        try {
            GlobalInstances.gsonOnlyExposed.fromJson(
                eventInfo,
                RCSServerMessageEvent::class.java
            )?.let {
                Log.i(TAG, "Received msg ${it.message}")
            }
        } catch (e: Exception) {
            e.printStackTrace()
        }
    }

}

private val mCheckConnectionCallback = object : TestDeviceManager.JobDoneCallback {
    override fun onJobDone(result: String?) {
        try {
            GlobalInstances.gsonAll.fromJson(result, CheckConnectionResult::class.java).let {
                Log.i(TAG, "Connection state: ${it.isConnected}")
                mIsConnected = it.isConnected
            }
        } catch (e: Exception) {
            e.printStackTrace()
        }
    }

    override fun onJobFailed(code: Int, reason: String?) {
        Log.e(TAG, "Check connection failed $reason")
    }
}

private fun CheckServerConnection() {
    val category = "remotecontrolJobs"
    val type = "IsServerConnected"
    TestDeviceManager.instance?.addJob(category, type, null, null, mCheckConnectionCallback)
}

private fun sendTestMessage() {
    val category = "remotecontrolJobs"
    val type = "SendMessageToServer"
    SendServerMessageParams().apply {
        message = "this is a test message ${messageCount++}"
    }.let {
        TestDeviceManager.instance?.addJob(category, type, it, null, null)
    }
}

private fun startReceiveServerMessage() {
    val category = "remotecontrolJobs"
    val type = "RegisterServerMessageListener"
    TestDeviceManager.instance?.addJob(category, type)
}

private fun startCheckConnection() {
    GlobalScope.launch {
        while (true) {
            CheckServerConnection()
            if (mIsConnected) {
                sendTestMessage()
            }
            delay(3000L)
        }
    }
}

private fun registerEventCallback() {
    TestDeviceManager.instance!!.setEventCallbacks(
        EVENT_NAME_SERVER_MESSAGE,
        mServerMessageReceivedHandler
    )
}

fun run() {
    registerEventCallback()
    startReceiveServerMessage()
    startCheckConnection()
}

}
