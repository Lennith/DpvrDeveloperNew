Sensor#

Here is an example of using Sensor, which listens to the distance of PSensor, and ends the listening after 1 minute.

class ApiTest {
    companion object TestStatic{
        const val TAG = "DMST-Sensor"
        const val PROXIMITY_SENSOR_DISTANCE_CHANGED = "proximity_sensor_distance_changed"
    }

    class PSensorDistanceChanged {
        @Expose
        var distance : Float = 0.0f
    }

    private val mSensorCallback = object : TestDeviceManager.EventCallback {
        override fun onEvent(eventInfo: String?) {
            try {
                val sensor = GlobalInstances.gsonOnlyExposed.fromJson(eventInfo,PSensorDistanceChanged::class.java)
                Log.i(TAG,"Proximity sensor distance ${sensor.distance}")
            } catch (e: Exception) {
                e.printStackTrace()
            }
        }
    }

    private fun setEventCallbacks() {
        TestDeviceManager.instance?.setEventCallbacks(
            PROXIMITY_SENSOR_DISTANCE_CHANGED,
            mSensorCallback
        )
    }

    private fun registerCallback() {
        val category = "sensorJobs"
        val type = "RegisterPSensorListener"
        TestDeviceManager.instance?.addJob(category, type, null, null, null)
    }

    private fun removeCallback(){
        val category = "sensorJobs"
        val type = "UnregisterPSensorListener"
        TestDeviceManager.instance?.addJob(category, type, null, null, null)
    }

    fun run() {
        registerCallback()
        setEventCallbacks()
        GlobalScope.launch {
            while(true){
                registerCallback()
                delay(60 * 1000)
                Log.i(TAG,"Remove sensor callback")
                removeCallback()
            }
        }
    }
}
