VIDEO_PLAY_STATUS#

DPVR standalone player’s playback status.

class VideoPlayStatusEvent {
    var time: Float? = 0.0f
    var videoName: String? = ""
    var videoPath: String? = ""
    var videoDisplayType: Int? = -1
    var state: String? = "STOPED"
    var duration: Float? = 0.0f
}

Permission#

APP_PERMISSION_NORMAL

Type name#

video_player_status

Property#

Float time: Video playback progress (in seconds).

String VideoName: Video filename.

String VideoPath: Video file path.

Int videoDisplayType: Video display type.

0: Normal 2D.

1: 3D side-by-side.

4: 3D top-bottom.

16: Panorama 2D.

17: Panorama 3D side-by-side.

20: Panorama 3D top-bottom.

32: 180 degrees 2D.

33: 180 degrees 3D side-by-side.

36: 180 degrees 3D top-bottom.

64: Fisheye 2D.

65: Fisheye 3D side-by-side.

68: Fisheye 3D top-bottom.

128: EAC 2D.

129: EAC 3D side-by-side.

1024: Cubemap.

String state: Video playback status.

“PREPARED”: Ready to play.

“BUFFERING”: Buffering.

“PLAYING”: Playing.

“PAUSE”: Paused.

“STOPED”: Stopped.

Float duration: Total video duration (in seconds).
