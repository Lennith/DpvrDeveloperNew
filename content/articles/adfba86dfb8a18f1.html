WifiDisplay#

Here is a demo of WifiDisplay, which scans for WifiDisplay devices, and connects to one that contains “Hagbis”, and then rescan after the connection is broken.

class ApiTest {
    companion object ApiTest{
        const val TAG = "DMST-WD"
    }

    class WifiDisplayDeviceInfo {
        @Expose
        var mac: String? = null

        @Expose
        var name: String? = null

        @Expose
        var isAvailable = false
        override fun toString(): String {
            return "WifiDisplay Info: $name, $mac, is active $isAvailable"
        }
    }

    class WifiDisplayScanResultEvent {
        companion object ScanResult{
            const val EVENT_TYPE = "wifi_display_scan_result"
        }

        @Expose
        var list: List<WifiDisplayDeviceInfo>? = null
    }

    class WifiDisplayConnectionChanged {
        companion object ConnectionChanged{
            const val EVENT_TYPE = "wifi_display_connection_changed"
        }

        @Expose
        var connected: Boolean = false

        @Expose
        var deviceInfo: WifiDisplayDeviceInfo? = null
    }

    class StartScanParams {
        var duration: Int = 10 * 1000
    }

    class ConnectParams {
        var name: String? = null
        var mac: String? = null
    }

    class ConnectResult {
        var device: com.dpvr.devicemanager.module.cast.WifiDisplayDeviceInfo? = null
        var success = false
        var failReason: String? = null
    }

    private val mScanResultCallback = object : TestDeviceManager.EventCallback {
        override fun onEvent(eventInfo: String?) {
            try {
                GlobalInstances.gsonOnlyExposed.fromJson(
                    eventInfo,
                    WifiDisplayScanResultEvent::class.java
                )?.let {
                    it.list?.let { deviceList ->
                        deviceList.forEach { device ->
                            Log.i(TAG, "Found device ${device.name}")
                            if (!device.name.isNullOrEmpty() && device.name!!.contains("HAGiBiS")) {
                                connectWifiDisplay(device.mac!!)
                            }
                        }
                    }
                }
            } catch (e: Exception) {
                e.printStackTrace()
            }
        }
    }

    private val mConnectionCallback = object : TestDeviceManager.EventCallback {
        override fun onEvent(eventInfo: String?) {
            try {
                GlobalInstances.gsonOnlyExposed.fromJson(
                    eventInfo,
                    WifiDisplayConnectionChanged::class.java
                )?.let {
                    if (it.connected) {
                        Log.i(TAG, "On connected to wifi display: ${it.deviceInfo?.name}")
                    } else {
                        Log.i(TAG, "On wifi display disconnected ${it.deviceInfo?.name}")
                        startScan()
                    }
                }
            } catch (e: Exception) {
                e.printStackTrace()
            }
        }
    }


    fun setEventCallbacks() {
        TestDeviceManager.instance?.setEventCallbacks(
            WifiDisplayConnectionChanged.EVENT_TYPE,
            mConnectionCallback
        )
        TestDeviceManager.instance?.setEventCallbacks(
            WifiDisplayScanResultEvent.EVENT_TYPE,
            mScanResultCallback
        )
    }

    private fun registerCallback() {
        val category = "wifidisplayJobs"
        val type = "RegisterConnectionCallback"
        TestDeviceManager.instance?.addJob(category, type, null, null, null)
    }

    private fun startScan() {
        val category = "wifidisplayJobs"
        val type = "StartScan"
        val params = StartScanParams().apply {
            duration = 600 * 1000
        }
        TestDeviceManager.instance?.addJob(category, type, params, null, null)
    }

    private fun connectWifiDisplay(mac: String) {
        val category = "wifidisplayJobs"
        val type = "ConnectWifiDisplay"
        val params = ConnectParams().apply {
            this.mac = mac
        }
        TestDeviceManager.instance?.addJob(
            category,
            type,
            params,
            null,
            object : TestDeviceManager.JobDoneCallback {
                override fun onJobDone(result: String?) {
                    Log.i(TAG, "On Connect job done, result $result")
                }

                override fun onJobFailed(code: Int, reason: String?) {
                    Log.i(TAG, "On Connect failed")
                }
            })
    }

    private fun stopWifiDisplay() {
        val category = "wifidisplayJobs"
        val type = "StopWifiDisplay"
        TestDeviceManager.instance?.addJob(category, type, null, null, null)
    }

    fun run() {
        registerCallback()
        setEventCallbacks()
        stopWifiDisplay()
        startScan()
    }
}
