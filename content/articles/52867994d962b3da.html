Messenger#

Here is an example of using Messenger that periodically sends messages to itself and also handles the reception of messages.

class ApiTest {
    companion object TestStatic{
        const val TAG = "DMST-MT"
        const val MESSENGER_RECEIVED = "messenger_received"
    }

    class MessageReceivedEvent {
        val type: DpvrDeviceManagerEvents = DpvrDeviceManagerEvents.MESSENGER_RECEIVED

        @Expose
        var fromPackage: String = ""

        @Expose
        var message: String? = null
    }

    class SendMessageParams(){
        var toPackage: String = ""
        var message: String? = null
    }

    private var mMessageCount = 0

    private val mMessageCallback = object : TestDeviceManager.EventCallback {
        override fun onEvent(eventInfo: String?) {
            try {
                val msg = GlobalInstances.gsonOnlyExposed.fromJson(eventInfo,MessageReceivedEvent::class.java)
                Log.i(TAG,"Received message from package: ${msg.fromPackage} : ${msg.message}")
            } catch (e: Exception) {
                e.printStackTrace()
            }
        }
    }


    private fun setEventCallbacks() {
        TestDeviceManager.instance?.setEventCallbacks(
            MESSENGER_RECEIVED,
            mMessageCallback
        )
    }

    private fun registerCallback() {
        val category = "messengerJobs"
        val type = "StartListenMessenger"
        TestDeviceManager.instance?.addJob(category, type, null, null, null)
    }

    private fun stopListenMessenger(){
        val category = "messengerJobs"
        val type = "StopListenMessenger"
        TestDeviceManager.instance?.addJob(category, type, null, null, null)
    }

    private fun sendTestMessage(){
        val category = "messengerJobs"
        val type = "SendMessage"
        val apiParams = SendMessageParams().apply {
            /** we send a test message to self */
            toPackage = GlobalInstances.serviceContext!!.packageName
            message = "this is ${mMessageCount++} message"
        }
        TestDeviceManager.instance?.addJob(category, type, apiParams, null, null)
    }

    fun run() {
        registerCallback()
        setEventCallbacks()
        GlobalScope.launch {
            while(true){
                sendTestMessage()
                delay(2000)
            }
        }
    }
}
