Examples#

We provide a few examples to demonstrate how to use the DeviceManager.

  • Messenger
  • Sensor
  • RemoteDeviceControl
  • WifiDisplay

All demonstrate are based on a DeviceManager caller class as below.

import android.annotation.SuppressLint
import android.app.Service
import android.content.ComponentName
import android.content.Context
import android.content.Intent
import android.content.ServiceConnection
import android.os.IBinder
import android.util.Log
import com.dpvr.devicemanager.IDeviceManager
import com.dpvr.devicemanager.IDeviceManagerEventListener
import com.dpvr.devicemanager.IDeviceManagerResultListener
import com.google.gson.Gson
import java.util.concurrent.LinkedBlockingDeque

class TestDeviceManager private constructor(private val mContext: Context) {
    interface JobDoneCallback {
        fun onJobDone(result: String?)

        fun onJobFailed(code: Int, reason: String?)
    }

    interface EventCallback {
        fun onEvent(eventInfo: String?)
    }

    internal inner class JobInfo {
        var cat: String? = null
        var type: String? = null
        var params: Any? = null
        var callParams: CallParams? = null
        var callback: JobDoneCallback? = null
    }

    class CallParams {
        var delay = 0
    }

    inner class ApiFinishedResult {
        var jobId = 0
        var returnCode = 0
        var result: String? = null
        var error: String? = null
    }

    internal inner class CallResult {
        var errorDescription: String? = null
        var jobID = 0
    }

    private val mJobCallbacks: HashMap<Int?, JobDoneCallback?> = HashMap()
    private val mEventCallbacks: HashMap<String?, EventCallback?> = HashMap()
    private val mRemainJobQueue: LinkedBlockingDeque<JobInfo?> = LinkedBlockingDeque()
    private val mGson: Gson = Gson()
    var gson: Gson = mGson
        get() = mGson

    private var mDeviceManager: IDeviceManager? = null
    private var mResultListener: IDeviceManagerResultListener? =
        object : IDeviceManagerResultListener.Stub() {
            override fun onResult(jobId: Int, result: String?) {
                val callback = mJobCallbacks[jobId]
                if (callback != null) {
                    val r: ApiFinishedResult = mGson.fromJson(result, ApiFinishedResult::class.java)
                    if (r.returnCode == 0) {
                        callback.onJobDone(r.result)
                    } else {
                        callback.onJobFailed(jobId, r.error)
                    }
                    mJobCallbacks.remove(jobId)
                }
            }
        }

    private var mEventListener: IDeviceManagerEventListener? =
        object : IDeviceManagerEventListener.Stub() {
            override fun onEvent(eventType: String?, eventInfo: String?) {
                if (mEventCallbacks[eventType] != null) {
                    mEventCallbacks[eventType]!!.onEvent(eventInfo)
                }
            }
        }
    private val connection: ServiceConnection = object : ServiceConnection {
        override fun onServiceConnected(componentName: ComponentName, iBinder: IBinder) {
            Log.d(TAG, "on service connected" + componentName.className)
            mDeviceManager = IDeviceManager.Stub.asInterface(iBinder)
            try {
                mDeviceManager!!.setDeviceManagerEventListener(mEventListener)
                mDeviceManager!!.setDeviceManagerResultListener(mResultListener)
                for (job in mRemainJobQueue) {
                    addJob(job!!.cat, job.type, job.params, job.callParams, job.callback)
                }
                mRemainJobQueue.clear()
            } catch (e: Exception) {
                e.printStackTrace()
            }
        }

        override fun onServiceDisconnected(componentName: ComponentName) {
            Log.e(TAG, "On dms service disconnect")
        }
    }

    init {
        bindService()
    }

    private fun bindService() {
        val intent = Intent("com.dpvr.devicemanager")
        intent.setPackage("com.dpvr.devicemanager")
        mContext.bindService(intent, connection, Service.BIND_AUTO_CREATE)
    }

    private fun unbindService() {
        mContext.unbindService(connection)
    }

    fun release() {
        unbindService()
        mJobCallbacks.clear()
        mEventCallbacks.clear()
        mEventListener = null
        mRemainJobQueue.clear()
        mResultListener = null
    }

    fun addJob(
        cat: String?,
        type: String?,
        params: Any?,
        callParams: CallParams?,
        callback: JobDoneCallback?
    ) {
        if (mDeviceManager == null) {
            val jobInfo: JobInfo = JobInfo()
            jobInfo.cat = cat
            jobInfo.params = params
            jobInfo.type = type
            jobInfo.callback = callback
            mRemainJobQueue.add(jobInfo)
        } else {
            try {
                val apiParams: String = if (params is String) params else mGson.toJson(params)
                val result: String =
                    mDeviceManager!!.call(cat, type, apiParams, mGson.toJson(callParams))
                val cl: CallResult = mGson.fromJson(result, CallResult::class.java)
                if (cl.jobID >= 0) {
                    mJobCallbacks[cl.jobID] = callback
                } else {
                    Log.e(TAG, "add job failed " + result + "reason:" + cl.errorDescription)
                }
            } catch (e: Exception) {
                e.printStackTrace()
            }
        }
    }

    fun setEventCallbacks(eventName: String?, callback: EventCallback) {
        mEventCallbacks.remove(eventName)
        mEventCallbacks[eventName] = callback
    }

    @SuppressLint("StaticFieldLeak")
    companion object me{
        private const val TAG = "QA-DMS"
        var instance: TestDeviceManager? = null
            private set

        @Synchronized
        fun createInstance(context: Context) {
            if (instance == null) {
                instance = TestDeviceManager(context)
            }
        }
    }
}
