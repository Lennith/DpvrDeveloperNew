Java Example Encapsulation#

Here, we use a simple encapsulation as an example for a developer to call DeviceManagerService.

class DeviceManager {
    public interface JobDoneCallback {
        void onJobDone(ApiFinishedResult result);
    }

    public interface EventCallback{
        void onEvent(String eventInfo);
    }

    class JobInfo {
        String cat = null;
        String type = null;
        Object params = null;
        String callParams = null;
        JobDoneCallback callback = null;
    }

    public static class CallParams{
        public int delay = 0;
    }

    public class ApiFinishedResult {
        public int jobId = 0;
        public int returnCode = 0;
        public String result = null;
        public String error = null;
    }

    class CallResult {
        String errorDescription = null;
        int jobID = 0;
    }

    private static final String TAG = "PS-DMS";
    private HashMap<Integer, JobDoneCallback> mJobCallbacks = new HashMap();
    private HashMap<String, EventCallback> mEventCallbacks = new HashMap();
    private LinkedBlockingDeque<JobInfo> mRemainJobQueue = new LinkedBlockingDeque();
    private Gson mGson = new Gson();
    private IDeviceManager mDeviceManager = null;

    private IDeviceManagerResultListener mResultListener =
        new IDeviceManagerResultListener.Stub() {
            @Override
            public void onResult(int jobId, String result) throws RemoteException {
                JobDoneCallback callback = mJobCallbacks.get(jobId);
                if(callback != null){
                    ApiFinishedResult r = mGson.fromJson(result, ApiFinishedResult.class);
                    callback.onJobDone(r);
                    mJobCallbacks.remove(jobId);
                }
            }
        };

    private IDeviceManagerEventListener mEventListener =
        new IDeviceManagerEventListener.Stub() {
            @Override
            public void onEvent(String eventType, String eventInfo)  {
                if(mEventCallbacks != null && mEventCallbacks.get(eventType) != null){
                    mEventCallbacks.get(eventType).onEvent(eventInfo);
                }
            }
        };

    private ServiceConnection connection = new ServiceConnection() {
        @Override
        public void onServiceConnected(ComponentName componentName, IBinder iBinder) {
            Log.d(TAG, "on service connected" + componentName.getClassName());
            mDeviceManager = IDeviceManager.Stub.asInterface(iBinder);
            try {
                mDeviceManager.setDeviceManagerEventListener(mEventListener);
                mDeviceManager.setDeviceManagerResultListener(mResultListener);
                for (JobInfo job : mRemainJobQueue) {
                    addJob(job.cat, job.type, job.params, job.callParams, job.callback);
                }
                mRemainJobQueue.clear();
            } catch (Exception e) {
                e.printStackTrace();
            }
        }

        @Override
        public void onServiceDisconnected(ComponentName componentName) {
            Log.e(TAG, "On dms service disconnect");
        }
    };

    private static DeviceManager _instance = null;
    private Context mContext ;

    private DeviceManager(Context context) {
        this.mContext = context;
        bindService();
    }

    private void bindService() {
        Intent intent = new Intent("com.dpvr.devicemanager");
        intent.setPackage("com.dpvr.devicemanager");
        mContext.bindService(intent, connection, Service.BIND_AUTO_CREATE);
    }

    private void unbindService() {
        if(connection != null) {
            mContext.unbindService(connection);
        }
    }

    public synchronized static void createInstance(Context context) {
        if (_instance == null) {
            _instance = new DeviceManager(context);
        }
    }

    public static DeviceManager getInstance() {
        return _instance;
    }

    public void release(){
        unbindService();
        mJobCallbacks.clear();
        mEventCallbacks.clear();
        mEventListener = null;
        mRemainJobQueue.clear();
        mResultListener = null;
    }

    public void addJob(String cat, String type, Object params, String callParams, JobDoneCallback callback) {
        if (mDeviceManager == null) {
            JobInfo jobInfo = new JobInfo();
            jobInfo.cat = cat;
            jobInfo.params = params;
            jobInfo.type = type;
            jobInfo.callback = callback;
            mRemainJobQueue.add(jobInfo);
        } else {
            try {
                String apiParams = mGson.toJson(params);
                String result = mDeviceManager.call(cat, type, apiParams, callParams);
                CallResult cl = mGson.fromJson(result, CallResult.class);
                if (cl.jobID >= 0) {
                    mJobCallbacks.put(cl.jobID, callback);
                } else {
                    Log.e(TAG, "add job failed " + result + "reason:" + cl.errorDescription);
                }
            } catch (Exception e) {
                e.printStackTrace();
            }
        }
    }

    public void setEventCallbacks(String eventName , EventCallback callback){
        mEventCallbacks.remove(eventName);
        mEventCallbacks.put(eventName,callback);
    }
}
